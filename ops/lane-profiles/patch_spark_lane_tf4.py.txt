#!/usr/bin/env python3
"""Register the TensorFold TP4 lane (dsv41-tf4) in spark-lane. Targeted, asserting patch; backup taken beforehand."""
import re, sys
P = "/Users/agent/recipe-db/spark-bench/ops/spark-lane"
s = open(P).read()

def sub1(old, new, count=1):
    global s
    assert s.count(old) == count, (old[:70], s.count(old))
    s = s.replace(old, new)

# 1. container names for dead-container cleanup
sub1('    dsv41-tp3|dsv41-tp4) echo "dsv41-head dsv41-worker" ;;\n',
     '    dsv41-tp3|dsv41-tp4) echo "dsv41-head dsv41-worker" ;;\n    dsv41-tf4)           echo "tp4-lane" ;;\n')

# 2. occupant detection: container name tp4-lane (recipe scripts/node_run.sh)
sub1("|glm53-tf3|glm53-tf4|^glm53[|]|^gr8[|]' || true)",
     "|glm53-tf3|glm53-tf4|^tp4-lane[|]|^glm53[|]|^gr8[|]' || true)")
sub1("  echo \"$out\" | grep -qiE 'dsv41-4x'           && lanes=\"$lanes dsv41-tp4\"\n",
     "  echo \"$out\" | grep -qiE 'dsv41-4x'           && lanes=\"$lanes dsv41-tp4\"\n"
     "  echo \"$out\" | grep -qE '^tp4-lane[|]'         && lanes=\"$lanes dsv41-tf4\"\n")

# 3. nodes + inventory
sub1('    dsv41-tp4)       echo "spark1 spark2 spark3 spark4" ;;\n',
     '    dsv41-tp4)       echo "spark1 spark2 spark3 spark4" ;;\n    dsv41-tf4)       echo "spark1 spark2 spark3 spark4" ;;\n')
sub1('ALL_LANE_IDS="comfy-engine dsv41-tp3 dsv41-tp4 glm53-tp3',
     'ALL_LANE_IDS="comfy-engine dsv41-tp3 dsv41-tp4 dsv41-tf4 glm53-tp3')

# 4. config + functions (inserted before dsv41_down)
cfg = '''
# ── dsv41-tf4: TensorFold TP4, LibertAI EXL3 3.5bpw pack (added 2026-10-24) ───────────────────────
# Source of truth: ~/recipe-db/DeepSeek-V4.1-Flash-TP2-and-TP4-DGX-Sparks (tonyd2wild) @ v0.3.5 (b88fae2),
# engine 0a0b69a + TF_RDMA_GID / TF_SEED_RANDOM patches. Our adapted config: models/dsv41-exl3-3.5bpw-tf/config.spark-fleet.env.
# Serves the SAME endpoint and model name as dsv41-tp4 (Hermes needs no change); context capped at 262144 (Victor, 2026-10-24).
TF4_ROOT="${TF4_ROOT:-/var/tmp/tf1017}"
TF4_CONFIG="${TF4_CONFIG:-$TF4_ROOT/config.spark-fleet.env}"
TF4_RECIPE="${TF4_RECIPE:-$TF4_ROOT/recipe}"
'''
fn = r'''
dsv41_tf4_up() {
  local lane="dsv41-tf4"
  STEPS=5; STEPNO=0; FAILED=0; NEXT_CMD="nothing"

  step "occupancy" "$(lane_nodes "$lane")"
  local conflict; conflict="$(occupancy_conflict "$lane")"
  if [[ -n "$conflict" ]]; then
    note_fail "occupancy" "nodes already taken:$conflict"
    NEXT_CMD="spark-lane rotate $lane   # parks the conflicting lanes, then brings this one up"
    result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"; return 1
  fi
  ok "all required nodes free"

  step "cleanup" "removing any dead tp4-lane containers"
  cleanup_dead_containers "$lane"
  ok "dead containers cleared (if any)"

  step "preflight" "serve.sh preflight on $DSV41_HEAD_HOST (every file rank 0 opens)"
  if ! timeout_run 120 ssh $SSH_OPTS "$DSV41_HEAD_HOST" "cd $TF4_RECIPE && DSV41_CONFIG=$TF4_CONFIG bash scripts/serve.sh preflight" >/dev/null 2>&1; then
    note_fail "preflight" "rank 0 is missing files (weights/cache/kit/engine) — nothing was started"
    NEXT_CMD="ssh $DSV41_HEAD_HOST 'cd $TF4_RECIPE && DSV41_CONFIG=$TF4_CONFIG bash scripts/serve.sh preflight'"
    result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"; return 1
  fi
  ok "preflight clean"

  step "start" "serve.sh start (ranks 3,2,1 then 0; waits for /v1/models and a 4-token chat)"
  if ! timeout_run 900 ssh $SSH_OPTS "$DSV41_HEAD_HOST" "cd $TF4_RECIPE && DSV41_CONFIG=$TF4_CONFIG bash scripts/serve.sh start" >/dev/null 2>&1; then
    note_fail "start" "serve.sh start failed or timed out (900s)"
    NEXT_CMD="ssh $DSV41_HEAD_HOST 'tail -40 $TF4_ROOT/logs/tp4-lane.log'   # then: spark-lane down $lane"
    result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"; return 1
  fi
  ok "ranks up, endpoint answering"

  step "identity" "served model name must be $DSV41_MODEL"
  if ! curl -s -m 10 "$DSV41_ENDPOINT/models" | grep -q "$DSV41_MODEL"; then
    note_fail "identity" "endpoint does not list $DSV41_MODEL"
    NEXT_CMD="curl -s $DSV41_ENDPOINT/models"
    result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"; return 1
  fi
  ok "serving $DSV41_MODEL"

  canary_tf2 "$DSV41_ENDPOINT" "$DSV41_MODEL" || { result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"; return 1; }
  result "$lane" "up" "$DSV41_ENDPOINT" "$DSV41_MODEL"
  return $FAILED
}

dsv41_tf4_down() {
  local lane="dsv41-tf4"
  STEPS=2; STEPNO=0; FAILED=0; NEXT_CMD="nothing"

  step "stop" "serve.sh stop (removes tp4-lane on all four nodes, bounded ${DSV41_STOP_TIMEOUT:-180}s)"
  if ! timeout_run "${DSV41_STOP_TIMEOUT:-180}" ssh $SSH_OPTS "$DSV41_HEAD_HOST" "cd $TF4_RECIPE && DSV41_CONFIG=$TF4_CONFIG bash scripts/serve.sh stop" >/dev/null 2>&1; then
    note_fail "stop" "serve.sh stop failed or timed out — verify the head node with 'ssh $DSV41_HEAD_HOST uptime'"
    NEXT_CMD="ssh $DSV41_HEAD_HOST 'cd $TF4_RECIPE && DSV41_CONFIG=$TF4_CONFIG bash scripts/serve.sh stop'"
    result "$lane" "down" "-" "-"; return 1
  fi
  ok "stop issued"

  step "verify torn down" "no tp4-lane containers left on $(lane_nodes "$lane")"
  local n left="" c
  for n in $(lane_nodes "$lane"); do
    c="$(docker_names "$n" | grep -E '^tp4-lane[|]' || true)"
    [[ -n "$c" ]] && left="$left $n:$c"
  done
  if [[ -n "$left" ]]; then
    note_fail "torn-down" "containers still present:$left"
    NEXT_CMD="spark-lane status   # inspect, then re-run down"
    result "$lane" "down" "-" "-"; return 1
  fi
  ok "all required nodes released"
  result "$lane" "down" "-" "-"
  NEXT_CMD="nothing — nodes released. Weights, rank caches and images were NOT deleted."
  return 0
}

'''
sub1("dsv41_down() {\n", cfg + fn + "dsv41_down() {\n")

# 5. dispatch
sub1("glm53-tf2) glm53_tf2_up ;; *) dsv41_up \"$LANE\"",
     "glm53-tf2) glm53_tf2_up ;; dsv41-tf4) dsv41_tf4_up ;; *) dsv41_up \"$LANE\"")
sub1("glm53-tf2) glm53_tf2_down ;; *) dsv41_down \"$LANE\"",
     "glm53-tf2) glm53_tf2_down ;; dsv41-tf4) dsv41_tf4_down ;; *) dsv41_down \"$LANE\"")

# 6. verify canary uses the reasoning-safe variant
sub1("    glm53-tf2|glm53-tp3|glm53-tp4|glm53-tp4-ablit) canary_tf2 \"$ep\" \"$model\" ;;",
     "    glm53-tf2|glm53-tp3|glm53-tp4|glm53-tp4-ablit|dsv41-tf4) canary_tf2 \"$ep\" \"$model\" ;;")

# 7. rotate park loop + logs + usage
sub1("for other in comfy-engine dsv41-tp3 dsv41-tp4 glm53-tp3",
     "for other in comfy-engine dsv41-tp3 dsv41-tp4 dsv41-tf4 glm53-tp3")
sub1('      dsv41-tp3|dsv41-tp4) kit="$(dsv41_kit "$LANE")"; host="$DSV41_HEAD_HOST" ;;\n',
     '      dsv41-tp3|dsv41-tp4) kit="$(dsv41_kit "$LANE")"; host="$DSV41_HEAD_HOST" ;;\n'
     '      dsv41-tf4) echo "${B}tf4 head log on $DSV41_HEAD_HOST${N} (tail ${TAIL:-40})"; sshx "$DSV41_HEAD_HOST" "tail -${TAIL:-40} $TF4_ROOT/logs/tp4-lane.log"; exit 0 ;;\n')
sub1("  dsv41-tp4         spark1-4      DeepSeek V4.1-Flash TP4 -> $DSV41_ENDPOINT  (fast line)\n",
     "  dsv41-tp4         spark1-4      DeepSeek V4.1-Flash TP4 -> $DSV41_ENDPOINT  (fast line)\n"
     "  dsv41-tf4         spark1-4      DeepSeek V4.1-Flash TP4 TensorFold, EXL3 3.5bpw -> $DSV41_ENDPOINT  (ctx 256K)\n")
sub1("#   dsv41-tp4         spark1-4      DeepSeek V4.1-Flash TP4  -> :8888/v1 (served as GLM-5.3-Flash-EXL3; Mia's fast line)\n",
     "#   dsv41-tp4         spark1-4      DeepSeek V4.1-Flash TP4  -> :8888/v1 (served as GLM-5.3-Flash-EXL3; Mia's fast line)\n"
     "#   dsv41-tf4         spark1-4      DeepSeek V4.1-Flash TP4 TensorFold (tonyd2wild recipe v0.3.5, LibertAI EXL3 3.5bpw) -> :8888/v1, same name, ctx 256K\n")

open(P, "w").write(s)
print("patched OK;", s.count("dsv41-tf4"), "mentions of dsv41-tf4")
